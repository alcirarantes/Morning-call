import 'dotenv/config'; import { writeFileSync, mkdirSync } from 'fs'; import { fetchMarketData } from '../lib/fetchMarket.js'; import { fetchBCBData } from '../lib/fetchBCB.js'; import { fetchFREDData } from '../lib/fetchFRED.js'; import { fetchCorporateNews, fetchEconomicCalendar } from '../lib/fetchNews.js'; import { generateMarketAnalysis } from '../lib/generateAnalysis.js'; import { buildHTML } from '../lib/buildHTML.js'; const isDev = process.argv.includes('--dev'); async function run() { const start = Date.now(); const timestamp = new Date().toISOString(); console.log(`\n🌅 Morning Call Generator`); console.log(`📅 ${new Date().toLocaleString('pt-BR', { timeZone: 'America/Sao_Paulo' })} (BRT)`); console.log(`─────────────────────────────────────`); // ── 1. Coleta paralela de todos os dados ────────────────────────────────── console.log('📊 Coletando dados de mercado...'); let market, bcb, fred, news, agenda; try { [market, bcb, fred, news, agenda] = await Promise.all([ fetchMarketData(), fetchBCBData(), fetchFREDData(process.env.FRED_API_KEY), fetchCorporateNews(), fetchEconomicCalendar(), ]); console.log(` ✅ Market: ${Object.keys(market).length} símbolos`); console.log(` ✅ BCB: Selic ${bcb.selic}%, USD/BRL ${bcb.cambio?.valor?.toFixed(4)}`); console.log(` ✅ FRED: Claims ${fred.initialClaims?.atual?.toLocaleString()}`); console.log(` ✅ News: ${news?.length ?? 0} manchetes`); } catch (e) { console.error('❌ Erro na coleta de dados:', e.message); process.exit(1); } // ── 2. Geração de texto analítico com Claude ────────────────────────────── console.log('🤖 Gerando análise com IA...'); let analysis = null; if (process.env.ANTHROPIC_API_KEY) { analysis = await generateMarketAnalysis( { market, bcb, fred, news, agenda }, process.env.ANTHROPIC_API_KEY ); if (analysis) { console.log(' ✅ Análise gerada com sucesso'); } else { console.warn(' ⚠️ Análise falhou — usando placeholders'); } } else { console.warn(' ⚠️ ANTHROPIC_API_KEY não configurada — pulando análise IA'); } // ── 3. Monta o HTML final ───────────────────────────────────────────────── console.log('🏗️ Montando HTML...'); const html = buildHTML({ market, bcb, fred, news, agenda, analysis }); // ── 4. Salva o arquivo ──────────────────────────────────────────────────── mkdirSync('./public', { recursive: true }); writeFileSync('./public/index.html', html, 'utf8'); const elapsed = ((Date.now() - start) / 1000).toFixed(1); console.log(`\n✅ Morning Call gerado em ${elapsed}s`); console.log(`📄 Arquivo: ./public/index.html`); if (isDev) { // Em modo dev, salva também um snapshot com timestamp const snapFile = `./public/snapshot-${Date.now()}.html`; writeFileSync(snapFile, html, 'utf8'); console.log(`🗂️ Snapshot: ${snapFile}`); } return { ok: true, timestamp, elapsed }; } // Executa se chamado diretamente run().catch(e => { console.error('❌ Falha crítica:', e); process.exit(1); }); export { run };
